# intervalClaude v0.2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade intervalClaude from long-lived setup-token to OAuth credential management — enables real-time usage queries, automatic token refresh, global account switch via Keychain swap, and interactive multi-account TUI dashboard.

**Architecture:** v0.2 maintains the same daemon/CLI architecture but swaps the data model: `account.token` → `account.credentials` (full OAuth payload). New modules `keychain.js` (macOS Keychain read/write via `security` command) and `oauth.js` (refresh token + query usage via `https://api.anthropic.com/api/oauth/usage`). Three new commands: `use` (atomic Keychain swap), `remove` (delete account), `tui` (interactive dashboard). Daemon adds usage polling + auto-refresh on each scan cycle.

**Tech Stack:** Node.js 18+ (ESM, zero runtime deps). Uses macOS `security` CLI for Keychain access via `child_process`. HTTP via Node's built-in `fetch` (Node 18+).

**Spec:** `docs/superpowers/specs/2026-05-23-v0.2-oauth-multiaccount-design.md`

**Critical Discovery:** Claude CLI's OAuth `client_id` is `9d1c250a-e61b-44d9-88ed-5944d1962f5e` (extracted from `strings $(which claude)`). This unblocks the token refresh flow.

// src/keychain.js
// macOS Keychain read/write for Claude CLI's "Claude Code-credentials"
//
// Reads/writes the full JSON payload that claude CLI stores.
// Linux/Windows: not supported in v0.2 — callers must check isKeychainSupported() first.

import { execFileSync } from 'node:child_process';

const KEYCHAIN_SERVICE = 'Claude Code-credentials';

export function isKeychainSupported() {
  return process.platform === 'darwin';
}

/**
 * Read raw JSON payload from macOS Keychain.
 * @returns {string|null} - raw JSON string, or null if no entry exists
 * @throws {Error} on unsupported platform
 */
export function readKeychainRaw() {
  if (!isKeychainSupported()) {
    throw new Error('Keychain only supported on macOS in v0.2');
  }
  try {
    const out = execFileSync('security', [
      'find-generic-password',
      '-s', KEYCHAIN_SERVICE,
      '-w',
    ], { encoding: 'utf8' });
    return out.trim();
  } catch (err) {
    // security exits non-zero when item not found
    if (err.status === 44 || /could not be found/i.test(String(err.stderr ?? ''))) {
      return null;
    }
    throw new Error(`Keychain read failed: ${err.message}`);
  }
}

/**
 * Write the full payload back to Keychain (creates or updates).
 * @param {string} rawJson - the full JSON payload to store
 */
export function writeKeychainRaw(rawJson) {
  if (!isKeychainSupported()) {
    throw new Error('Keychain only supported on macOS in v0.2');
  }
  // -U updates if exists, otherwise creates
  execFileSync('security', [
    'add-generic-password',
    '-U',
    '-s', KEYCHAIN_SERVICE,
    '-a', process.env.USER || 'claude',
    '-w', rawJson,
  ], { stdio: 'ignore' });
}

/**
 * Parse the JSON payload and extract Claude OAuth credentials.
 * @param {string} rawJson
 * @returns {Object} - {accessToken, refreshToken, expiresAt, scopes, subscriptionType, rateLimitTier}
 * @throws {Error} if JSON is invalid or claudeAiOauth field is missing
 */
export function parseClaudeCredentials(rawJson) {
  let parsed;
  try {
    parsed = JSON.parse(rawJson);
  } catch (err) {
    throw new Error(`Invalid Keychain JSON: ${err.message}`);
  }
  const oauth = parsed.claudeAiOauth;
  if (!oauth || !oauth.accessToken) {
    throw new Error('Keychain payload missing claudeAiOauth field');
  }
  return {
    accessToken: oauth.accessToken,
    refreshToken: oauth.refreshToken,
    expiresAt: oauth.expiresAt,
    scopes: oauth.scopes || [],
    subscriptionType: oauth.subscriptionType,
    rateLimitTier: oauth.rateLimitTier,
  };
}

/**
 * Serialize credentials back to the Keychain JSON shape (preserving mcpOAuth from original if available).
 * @param {Object} credentials - {accessToken, refreshToken, expiresAt, scopes, subscriptionType, rateLimitTier}
 * @param {string|null} originalRaw - the current Keychain raw JSON (for preserving mcpOAuth), or null
 * @returns {string} - JSON string ready for writeKeychainRaw
 */
export function serializeCredentials(credentials, originalRaw = null) {
  let preserved = {};
  if (originalRaw) {
    try {
      const parsed = JSON.parse(originalRaw);
      if (parsed.mcpOAuth) preserved.mcpOAuth = parsed.mcpOAuth;
    } catch {
      // ignore — fall through to no preservation
    }
  }
  return JSON.stringify({
    claudeAiOauth: {
      accessToken: credentials.accessToken,
      refreshToken: credentials.refreshToken,
      expiresAt: credentials.expiresAt,
      scopes: credentials.scopes,
      subscriptionType: credentials.subscriptionType,
      rateLimitTier: credentials.rateLimitTier,
    },
    ...preserved,
  });
}

/**
 * Convenience: read + parse in one call.
 * @returns {Object|null} - parsed credentials, or null if no entry
 */
export function readCredentials() {
  const raw = readKeychainRaw();
  if (raw === null) return null;
  return parseClaudeCredentials(raw);
}
```

// src/oauth.js
// OAuth operations: token refresh + usage query
//
// Claude CLI's OAuth client_id is hardcoded in the CLI binary.
// Refresh and usage endpoints are part of the unofficial-but-stable
// Anthropic API used by claude CLI itself.

export const CLIENT_ID = '9d1c250a-e61b-44d9-88ed-5944d1962f5e';
export const USAGE_ENDPOINT = 'https://api.anthropic.com/api/oauth/usage';
export const TOKEN_ENDPOINT = 'https://api.anthropic.com/api/oauth/token';
export const OAUTH_BETA_HEADER = 'oauth-2025-04-20';

const DEFAULT_EXPIRY_THRESHOLD_MS = 10 * 60 * 1000; // 10 minutes

/**
 * @param {Object} credentials - {expiresAt}
 * @param {number} thresholdMs - default 10 min
 * @returns {boolean} - true if expired or within threshold of expiring
 */
export function isExpiringSoon(credentials, thresholdMs = DEFAULT_EXPIRY_THRESHOLD_MS) {
  if (!credentials || !credentials.expiresAt) return true;
  return credentials.expiresAt - Date.now() < thresholdMs;
}

/**
 * Refresh an OAuth token using its refresh_token.
 * @param {Object} credentials - {refreshToken, ...other fields preserved}
 * @param {Object} options - {fetchFn: global fetch, now: Date}
 * @returns {Promise<Object>} - new credentials with rotated tokens
 * @throws {Error} on network error or refresh rejection
 */
export async function refreshAccessToken(credentials, options = {}) {
  const fetchFn = options.fetchFn ?? fetch;
  if (!credentials.refreshToken) {
    throw new Error('Missing refreshToken — cannot refresh');
  }
  const res = await fetchFn(TOKEN_ENDPOINT, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'anthropic-beta': OAUTH_BETA_HEADER,
    },
    body: JSON.stringify({
      grant_type: 'refresh_token',
      refresh_token: credentials.refreshToken,
      client_id: CLIENT_ID,
    }),
  });
  if (!res.ok) {
    const body = await res.text();
    throw new Error(`Token refresh failed (${res.status}): ${body.slice(0, 200)}`);
  }
  const data = await res.json();
  return {
    ...credentials,
    accessToken: data.access_token,
    refreshToken: data.refresh_token || credentials.refreshToken,
    expiresAt: Date.now() + (data.expires_in || 28800) * 1000,
  };
}

/**
 * Query usage data from Anthropic API.
 * @param {string} accessToken - OAuth access token with user:profile scope
 * @param {Object} options - {fetchFn: global fetch}
 * @returns {Promise<Object>} - {five_hour: {utilization, resets_at}, seven_day: {...}, ...}
 * @throws {Error} on network error or auth failure
 */
export async function queryUsage(accessToken, options = {}) {
  const fetchFn = options.fetchFn ?? fetch;
  const res = await fetchFn(USAGE_ENDPOINT, {
    method: 'GET',
    headers: {
      Authorization: `Bearer ${accessToken}`,
      'anthropic-beta': OAUTH_BETA_HEADER,
      Accept: 'application/json',
    },
  });
  if (res.status === 401) {
    const body = await res.text();
    throw Object.assign(new Error(`Usage query unauthorized: ${body.slice(0, 200)}`), {
      code: 'UNAUTHORIZED',
    });
  }
  if (!res.ok) {
    const body = await res.text();
    throw new Error(`Usage query failed (${res.status}): ${body.slice(0, 200)}`);
  }
  const data = await res.json();
  // Some fields may be returned as percentage (0-100) or fraction (0-1).
  // The actual API returns percentages (e.g. 57.0). Normalize to fraction.
  return {
    five_hour: data.five_hour ? {
      utilization: normalizeUtilization(data.five_hour.utilization),
      resets_at: data.five_hour.resets_at,
    } : null,
    seven_day: data.seven_day ? {
      utilization: normalizeUtilization(data.seven_day.utilization),
      resets_at: data.seven_day.resets_at,
    } : null,
    seven_day_opus: data.seven_day_opus ? {
      utilization: normalizeUtilization(data.seven_day_opus.utilization),
      resets_at: data.seven_day_opus.resets_at,
    } : null,
    seven_day_sonnet: data.seven_day_sonnet ? {
      utilization: normalizeUtilization(data.seven_day_sonnet.utilization),
      resets_at: data.seven_day_sonnet.resets_at,
    } : null,
    extra_usage: data.extra_usage ?? null,
    fetched_at: new Date().toISOString(),
  };
}

/** Normalize utilization to a 0..1 fraction. Accepts 0..1 or 0..100. */
function normalizeUtilization(v) {
  if (v == null) return null;
  return v > 1 ? v / 100 : v;
}

/**
 * Query usage with automatic refresh on 401.
 * @param {Object} credentials
 * @param {Object} options
 * @returns {Promise<{usage, credentials}>} - usage data + possibly-refreshed credentials
 */
export async function queryUsageWithRefresh(credentials, options = {}) {
  try {
    const usage = await queryUsage(credentials.accessToken, options);
    return { usage, credentials };
  } catch (err) {
    if (err.code !== 'UNAUTHORIZED') throw err;
    const refreshed = await refreshAccessToken(credentials, options);
    const usage = await queryUsage(refreshed.accessToken, options);
    return { usage, credentials: refreshed };
  }
}
```

// src/commands/use.js
// `use <name>` — atomically swap macOS Keychain to a different account.
//
// Workflow:
//   1. Read current Keychain → identify which account it belongs to → back it up
//   2. Refresh target account's token if expiring
//   3. Write target credentials to Keychain
//   4. Query target usage to verify + cache

import {
  isKeychainSupported,
  readKeychainRaw,
  writeKeychainRaw,
  parseClaudeCredentials,
  serializeCredentials,
} from '../keychain.js';
import { loadConfig, saveConfig, setCredentials, setLastUsage } from '../config.js';
import { refreshAccessToken, isExpiringSoon, queryUsage } from '../oauth.js';

export function findAccountByAccessToken(config, accessToken) {
  for (const a of config.accounts) {
    if (a.credentials?.accessToken === accessToken) return a;
    if (a.legacy_token === accessToken) return a;
  }
  return null;
}

export default async function useCommand(args) {
  if (!isKeychainSupported()) {
    console.error('use: 仅支持 macOS');
    process.exit(1);
  }

  const targetName = args[0];
  if (!targetName) {
    console.error('用法: interval-claude use <name>');
    process.exit(1);
  }

  let config = await loadConfig();
  const target = config.accounts.find(a => a.name === targetName);
  if (!target) {
    console.error(`帐号 "${targetName}" 不存在`);
    process.exit(1);
  }
  if (!target.credentials) {
    console.error(`帐号 "${targetName}" 没有 OAuth 凭证（可能是 v0.1 长效 token）。请用 add 重新捕获。`);
    process.exit(1);
  }

  // Step 1: backup current Keychain
  const currentRaw = readKeychainRaw();
  if (currentRaw) {
    try {
      const currentParsed = parseClaudeCredentials(currentRaw);
      const owner = findAccountByAccessToken(config, currentParsed.accessToken);
      if (owner) {
        console.log(`备份当前 Keychain 到帐号: ${owner.name}`);
        config = setCredentials(config, owner.name, currentParsed);
      } else {
        console.warn('⚠️  当前 Keychain 中的 token 不属于已配置帐号，已忽略备份');
      }
    } catch (err) {
      console.warn(`⚠️  解析当前 Keychain 失败: ${err.message}`);
    }
  }

  // Step 2: refresh if expiring
  let targetCreds = target.credentials;
  if (isExpiringSoon(targetCreds)) {
    console.log(`目标 token 即将过期，正在刷新...`);
    try {
      targetCreds = await refreshAccessToken(targetCreds);
      config = setCredentials(config, targetName, targetCreds);
    } catch (err) {
      console.error(`token 刷新失败: ${err.message}`);
      console.error('请运行: claude /logout && claude /login（用 ' + targetName + ' 帐号），然后 interval-claude add ' + targetName + ' 重新捕获');
      process.exit(1);
    }
  }

  // Step 3: write Keychain
  const newRaw = serializeCredentials(targetCreds, currentRaw);
  writeKeychainRaw(newRaw);

  // Step 4: query usage
  let usage = null;
  try {
    usage = await queryUsage(targetCreds.accessToken);
    config = setLastUsage(config, targetName, usage);
  } catch (err) {
    console.warn(`⚠️  usage 查询失败: ${err.message}`);
  }

  await saveConfig(config);

  console.log(`✅ 已切换到 ${targetName}`);
  console.log(`   订阅: ${targetCreds.subscriptionType || '未知'}`);
  if (usage?.five_hour) {
    const pct = Math.round(usage.five_hour.utilization * 100);
    console.log(`   5h 使用: ${pct}% (剩余 ${100 - pct}%)`);
  }
  console.log('所有终端的 claude 命令立即生效');
}

function formatTable(config, activeAccessToken) {
  const rows = [
    ['NAME', 'SUB', '5H USE', 'RESETS', '7D USE', 'STATUS'],
  ];
  for (const a of config.accounts) {
    const isActive = activeAccessToken && getAccessToken(a) === activeAccessToken;
    const name = (isActive ? '* ' : '  ') + a.name;
    const sub = a.credentials?.subscriptionType ?? (a.legacy_token ? 'v0.1 token' : '-');
    const fiveH = formatUtilization(a.last_usage?.five_hour);
    const resets = formatTimeUntil(a.last_usage?.five_hour?.resets_at);
    const sevenD = formatUtilization(a.last_usage?.seven_day);
    const status = formatStatus(a.last_usage?.five_hour);
    rows.push([name, sub, fiveH, resets, sevenD, status]);
  }

  // Column widths
  const widths = rows[0].map((_, i) => Math.max(...rows.map(r => visibleWidth(r[i]))));
  const sep = widths.map(w => '-'.repeat(w)).join('  ');
  const lines = [];
  lines.push(rows[0].map((c, i) => pad(c, widths[i])).join('  '));
  lines.push(sep);
  for (let i = 1; i < rows.length; i++) {
    lines.push(rows[i].map((c, i) => pad(c, widths[i])).join('  '));
  }
  return lines.join('\n');
}

function formatUtilization(u) {
  if (!u || u.utilization == null) return '-';
  const pct = Math.round(u.utilization * 100);
  return `${pct}%`;
}

function formatTimeUntil(iso) {
  if (!iso) return '-';
  const ms = new Date(iso).getTime() - Date.now();
  if (ms <= 0) return 'now';
  const h = Math.floor(ms / 3600000);
  const m = Math.floor((ms % 3600000) / 60000);
  return h > 0 ? `in ${h}h${m}m` : `in ${m}m`;
}

function formatStatus(fiveH) {
  if (!fiveH || fiveH.utilization == null) return '⚪';
  const u = fiveH.utilization;
  if (u >= 0.95) return '🔴';
  if (u >= 0.80) return '🟡';
  return '🟢';
}

function visibleWidth(s) {
  // crude: count emoji/CJK as 2
  let w = 0;
  for (const ch of s) {
    w += ch.codePointAt(0) > 0x2e00 ? 2 : 1;
  }
  return w;
}

function pad(s, w) {
  return s + ' '.repeat(Math.max(0, w - visibleWidth(s)));
}

// Interactive multi-account dashboard.
//
// Refresh: every 5s from cached last_usage (no API call)
// Keys: r=force refresh API  u=switch account  p=ping  q=quit  s=start daemon (if not running)

import { loadConfig, saveConfig, getAccessToken, setLastUsage, setCredentials } from '../config.js';
import { daemonStatus, startDaemon } from '../daemon.js';
import { isKeychainSupported, readKeychainRaw, parseClaudeCredentials } from '../keychain.js';
import { queryUsageWithRefresh } from '../oauth.js';
import useCommand from './use.js';
import pingCommand from './ping-cmd.js';

const REFRESH_INTERVAL_MS = 5000;

export default async function tuiCommand() {
  process.stdout.write('\x1b[?25l'); // hide cursor

  // Set raw mode for single-key input
  if (process.stdin.isTTY) {
    process.stdin.setRawMode(true);
    process.stdin.resume();
    process.stdin.setEncoding('utf8');
  }

  let busy = false;
  process.stdin.on('data', async (key) => {
    if (busy) return;
    if (key === 'q' || key === '') return quit();
    if (key === 'r') return forceRefresh();
    if (key === 'u') return interactiveSwitch();
    if (key === 'p') return interactivePing();
    if (key === 's') return startInteractive();
  });

  process.on('SIGINT', quit);
  process.on('SIGTERM', quit);

  // Render loop
  const timer = setInterval(render, REFRESH_INTERVAL_MS);
  await render();

  function quit() {
    clearInterval(timer);
    process.stdout.write('\x1b[?25h\n'); // show cursor + newline
    if (process.stdin.isTTY) process.stdin.setRawMode(false);
    process.exit(0);
  }

  async function forceRefresh() {
    busy = true;
    let config = await loadConfig();
    for (const a of config.accounts) {
      if (!a.credentials) continue;
      try {
        const { usage, credentials } = await queryUsageWithRefresh(a.credentials);
        a.credentials = credentials;
        a.last_usage = usage;
      } catch { /* ignore individual failures */ }
    }
    await saveConfig(config);
    busy = false;
    render();
  }

  async function interactiveSwitch() {
    const config = await loadConfig();
    const name = await pickAccount(config, '切换到:');
    if (!name) return render();
    busy = true;
    try {
      await useCommand([name]);
    } catch (err) {
      // already printed
    }
    busy = false;
    render();
  }

  async function interactivePing() {
    const config = await loadConfig();
    const name = await pickAccount(config, '手动 ping:');
    if (!name) return render();
    busy = true;
    try {
      await pingCommand([name]);
    } catch { /* ignore */ }
    busy = false;
    render();
  }

  async function startInteractive() {
    busy = true;
    await startDaemon();
    busy = false;
    render();
  }
}

async function pickAccount(config, title) {
  // Temporarily disable raw mode for number input
  if (process.stdin.isTTY) process.stdin.setRawMode(false);
  process.stdout.write('\x1b[2J\x1b[H'); // clear
  console.log(title);
  config.accounts.forEach((a, i) => console.log(`  ${i + 1}. ${a.name}`));
  process.stdout.write('选择 (空白取消): ');
  const choice = await new Promise(resolve => {
    process.stdin.once('data', data => resolve(String(data).trim()));
  });
  if (process.stdin.isTTY) process.stdin.setRawMode(true);
  const idx = parseInt(choice, 10) - 1;
  return config.accounts[idx]?.name ?? null;
}

async function render() {
  const config = await loadConfig();
  const status = await daemonStatus();

  let activeAccessToken = null;
  if (isKeychainSupported()) {
    try {
      const raw = readKeychainRaw();
      if (raw) activeAccessToken = parseClaudeCredentials(raw).accessToken;
    } catch { /* ignore */ }
  }

  const headerText = status.running
    ? `intervalClaude · 守护进程运行中 (uptime ${status.uptime})`
    : 'intervalClaude · 守护进程未运行 — 按 [s] 启动';

  const lines = [];
  lines.push(`┌─ ${headerText} ─────────────────────────────────────┐`);
  lines.push('│                                                                       │');

  // Table header
  lines.push(`│  ${pad('NAME', 12)} ${pad('SUB', 8)} ${pad('5H USE', 10)} ${pad('7D USE', 8)} ${pad('NEXT PING', 12)} ${pad('RESETS', 12)} │`);
  lines.push(`│  ${'─'.repeat(64)}    │`);

  for (const a of config.accounts) {
    const isActive = activeAccessToken && getAccessToken(a) === activeAccessToken;
    const namePrefix = isActive ? '* ' : '  ';
    const name = pad(namePrefix + a.name, 12);
    const sub = pad(a.credentials?.subscriptionType ?? (a.legacy_token ? 'v0.1' : '-'), 8);
    const fiveH = pad(fmtUsageBar(a.last_usage?.five_hour), 10);
    const sevenD = pad(fmtUtil(a.last_usage?.seven_day), 8);
    const nextPing = pad(fmtNextPing(a, status), 12);
    const resets = pad(fmtTimeUntil(a.last_usage?.five_hour?.resets_at), 12);
    lines.push(`│  ${name} ${sub} ${fiveH} ${sevenD} ${nextPing} ${resets} │`);
  }

  lines.push('│                                                                       │');
  const activeName = config.accounts.find(a => activeAccessToken && getAccessToken(a) === activeAccessToken)?.name ?? '(未知)';
  lines.push(`│  当前 Keychain 活跃: ${pad(activeName, 50)} │`);
  lines.push('│                                                                       │');

  if (status.running) {
    lines.push('│  [r] 刷新 usage   [u] 切换帐号   [p] 手动 ping   [q] 退出           │');
  } else {
    lines.push('│  [s] 启动守护进程   [u] 切换帐号   [q] 退出                         │');
  }
  lines.push('└───────────────────────────────────────────────────────────────────────┘');

  // Clear + render
  process.stdout.write('\x1b[2J\x1b[H');
  process.stdout.write(lines.join('\n') + '\n');
}

function fmtUsageBar(u) {
  if (!u || u.utilization == null) return '-';
  const pct = Math.round(u.utilization * 100);
  const filled = Math.round(u.utilization * 5);
  return '█'.repeat(filled) + '░'.repeat(5 - filled) + ` ${pct}%`;
}

function fmtUtil(u) {
  if (!u || u.utilization == null) return '-';
  return Math.round(u.utilization * 100) + '%';
}

function fmtNextPing(account, status) {
  if (!status.running || !status.startedAt) return '-';
  const startedMs = new Date(status.startedAt).getTime();
  const lastPing = account.last_usage?.fetched_at;
  if (lastPing) {
    // Approximate: next ping = lastPing + interval * accounts.length
    return '~周期内';
  }
  const targetMs = startedMs + account.offset_minutes * 60000;
  const diff = targetMs - Date.now();
  if (diff <= 0) return '← 现在';
  const h = Math.floor(diff / 3600000);
  const m = Math.floor((diff % 3600000) / 60000);
  return h > 0 ? `in ${h}h${m}m` : `in ${m}m`;
}

function fmtTimeUntil(iso) {
  if (!iso) return '-';
  const ms = new Date(iso).getTime() - Date.now();
  if (ms <= 0) return 'now';
  const h = Math.floor(ms / 3600000);
  const m = Math.floor((ms % 3600000) / 60000);
  return h > 0 ? `in ${h}h${m}m` : `in ${m}m`;
}

function pad(s, w) {
  // crude width (emoji/CJK = 2)
  let cur = 0;
  for (const ch of s) cur += ch.codePointAt(0) > 0x2e00 ? 2 : 1;
  return s + ' '.repeat(Math.max(0, w - cur));
}
