# intervalClaude v0.3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or executing-plans. Steps use checkbox.

**Goal:** Implement dynamic stagger scheduling — daemon evaluates account health every cycle and auto-pings/switches based on usage, time, and sub tier.

**Architecture:** Replace v0.2's static i × 300/N pre-ping schedule with reactive decision loop. Each 60s cycle calls `schedule()` which returns either no-op, [PING], [USE], or [PING + USE] actions. Health formula combines sub weight × remaining usage × window remaining time.

**Tech Stack:** Node.js 18+ ESM, zero deps. New module `notifier.js` shells out to `osascript`. New module `scheduler.js` (replaces ad-hoc logic).

**Spec:** `docs/superpowers/specs/2026-05-24-v0.3-dynamic-scheduler-design.md`

```js
// src/scheduler.js
// v0.3 动态调度算法
//
// 输入：accounts 数组、当前活跃帐号、配置、当前时间
// 输出：动作数组，每个动作是 { type: 'PING'|'USE', account }
//
// 算法核心：每周期评估"目标状态"vs"实际状态"，差异触发动作。

export const DEFAULT_SUB_WEIGHTS = {
  pro: 1,
  max_5x: 5,
  max_20x: 20,
  team: 10,
  _default: 1,
};

export const DEFAULT_EXPIRE_THRESHOLD_MIN = 3;
export const FULL_WINDOW_MIN = 300; // 5h
export const LIMIT_REACHED_VALID_MS = 10 * 60 * 1000; // 10 min

// 动作类型
export const ACTION_PING = 'PING';
export const ACTION_USE = 'USE';

/**
 * 获取帐号 sub 权重
 */
function subWeight(account, cfg) {
  const weights = cfg?.scheduler?.sub_weights ?? DEFAULT_SUB_WEIGHTS;
  const sub = account?.credentials?.subscriptionType ?? '_default';
  return weights[sub] ?? weights._default ?? 1;
}

/**
 * 帐号窗口是否未启动
 */
function isWindowNotStarted(account) {
  return !account?.last_usage?.five_hour?.resets_at;
}

/**
 * 帐号窗口是否已过期
 */
function isWindowExpired(account, now) {
  const ts = account?.last_usage?.five_hour?.resets_at;
  if (!ts) return false;
  return new Date(ts).getTime() <= now;
}

/**
 * 窗口剩余分钟数（已激活且未过期才有意义）
 */
function windowRemainingMin(account, now) {
  const ts = account?.last_usage?.five_hour?.resets_at;
  if (!ts) return 0;
  const ms = new Date(ts).getTime() - now;
  return Math.max(0, ms / 60000);
}

/**
 * 当前 usage（0~1，未激活帐号返回 0）
 */
function currentUsage(account) {
  return account?.last_usage?.five_hour?.utilization ?? 0;
}

/**
 * 计算健康度评分
 */
export function health(account, cfg, now = Date.now()) {
  if (!account) return 0;
  const weight = subWeight(account, cfg);

  if (isWindowNotStarted(account)) {
    return weight * 1.0 * FULL_WINDOW_MIN;
  }
  if (currentUsage(account) >= 1.0) return 0;
  if (isWindowExpired(account, now)) {
    return weight * 1.0 * FULL_WINDOW_MIN;
  }
  const remaining = windowRemainingMin(account, now);
  const threshold = cfg?.scheduler?.expire_threshold_min ?? DEFAULT_EXPIRE_THRESHOLD_MIN;
  if (remaining < threshold) return 0;
  const remainingUsage = 1 - currentUsage(account);
  return weight * remainingUsage * remaining;
}

/**
 * 判断是否需要切换主帐号
 */
export function needsSwitch(active, cfg, now = Date.now()) {
  if (!active) return true;
  if (currentUsage(active) >= 1.0) return true;
  if (isWindowExpired(active, now)) return true;
  if (active.limit_reached_at &&
      now - active.limit_reached_at < LIMIT_REACHED_VALID_MS) {
    return true;
  }
  return false;
}

/**
 * 选最佳切换候选（健康度最高，排除自己和健康度=0 的）
 */
export function bestSwitchCandidate(accounts, exclude, cfg, now = Date.now()) {
  const candidates = accounts.filter((a) => {
    if (exclude && a.name === exclude.name) return false;
    return health(a, cfg, now) > 0;
  });
  if (candidates.length === 0) return null;
  candidates.sort((a, b) => health(b, cfg, now) - health(a, cfg, now));
  return candidates[0];
}

/**
 * 错峰预 ping 判断
 */
export function shouldPrePing(accounts, active, cfg, now = Date.now()) {
  if (!active) return null;
  const N = accounts.length;
  if (N <= 1) return null;

  // 未激活的候选（按健康度排序）
  const dormant = accounts
    .filter((a) => a.name !== active.name && isWindowNotStarted(a))
    .sort((a, b) => health(b, cfg, now) - health(a, cfg, now));
  if (dormant.length === 0) return null;

  // 已在跑的备用数（不含活跃）
  const runningBackups = accounts.filter((a) =>
    a.name !== active.name &&
    !isWindowNotStarted(a) &&
    !isWindowExpired(a, now)
  ).length;

  // 下一个该 ping 的位置（0-based）
  const nextIndex = runningBackups;
  if (nextIndex >= dormant.length) return null;

  const stagger = cfg?.scheduler?.stagger_min ?? (FULL_WINDOW_MIN / N);
  const targetTimeMin = (nextIndex + 1) * stagger;
  const targetUsageFrac = (nextIndex + 1) / N;

  // active.window_start 优先用 explicit 字段；否则从 last_usage.resets_at 反推（- 300min）
  const windowStartMs = active.window_start ??
    (active.last_usage?.five_hour?.resets_at
      ? new Date(active.last_usage.five_hour.resets_at).getTime() - FULL_WINDOW_MIN * 60_000
      : now);
  const elapsedMin = (now - windowStartMs) / 60_000;
  const usage = currentUsage(active);

  if (elapsedMin >= targetTimeMin || usage >= targetUsageFrac) {
    return dormant[0];
  }
  return null;
}

/**
 * 完整调度循环。
 */
export function schedule(accounts, active, cfg, now = Date.now()) {
  // Step 1: 切换决策
  if (needsSwitch(active, cfg, now)) {
    const candidate = bestSwitchCandidate(accounts, active, cfg, now);
    if (!candidate) return []; // 全耗尽
    if (isWindowNotStarted(candidate)) {
      return [
        { type: ACTION_PING, account: candidate },
        { type: ACTION_USE, account: candidate },
      ];
    }
    return [{ type: ACTION_USE, account: candidate }];
  }

  // Step 2: 预 ping 决策
  const next = shouldPrePing(accounts, active, cfg, now);
  if (next) {
    return [{ type: ACTION_PING, account: next }];
  }

  return [];
}
```

```js
// 在 daemon.js 顶部：
import { schedule, ACTION_PING, ACTION_USE } from './scheduler.js';
import { sendNotification } from './notifier.js';
import { isKeychainSupported, readKeychainRaw, parseClaudeCredentials } from './keychain.js';

// 替换 runDaemon 主循环：
export async function runDaemon(options = {}) {
  const checkIntervalMs = options.checkIntervalMs ?? DEFAULT_CHECK_INTERVAL_MS;
  const logFn = options.logFn ?? ((msg) => appendLog(msg));
  const pingFn = options.pingFn ?? pingWithRetry;
  const nowFn = options.nowFn ?? (() => new Date());
  const shouldStop = options.shouldStop ?? (() => false);

  logFn('daemon: 主循环启动 (v0.3 动态调度)');

  while (!shouldStop()) {
    let config;
    try {
      config = await loadConfig();
    } catch (err) {
      logFn(`daemon: load config 失败: ${err?.message ?? err}`);
      await interruptibleSleep(checkIntervalMs, shouldStop);
      continue;
    }

    if (!config.scheduler?.enabled) {
      await interruptibleSleep(checkIntervalMs, shouldStop);
      continue;
    }

    // 识别当前活跃帐号（通过 Keychain）
    let active = null;
    if (isKeychainSupported()) {
      try {
        const raw = readKeychainRaw();
        if (raw) {
          const creds = parseClaudeCredentials(raw);
          active = config.accounts.find(
            (a) => a.credentials?.accessToken === creds.accessToken
          );
        }
      } catch { /* ignore */ }
    }

    const now = nowFn().getTime();
    let actions;
    try {
      actions = schedule(config.accounts, active, config, now);
    } catch (err) {
      logFn(`daemon: schedule() 失败: ${err?.message ?? err}`);
      await interruptibleSleep(checkIntervalMs, shouldStop);
      continue;
    }

    if (actions.length === 0) {
      // 全满检测
      if (!active || needsSwitch(active, config, now)) {
        const hasAnyValid = config.accounts.some(
          (a) => !isWindowNotStarted(a) && currentUsage(a) < 1.0 && !isWindowExpired(a, now)
        );
        if (!hasAnyValid) {
          // 全满 → 找最早重置
          const earliest = findEarliestReset(config.accounts);
          if (earliest && !state.allExhaustedNotified) {
            await sendNotification({
              title: 'intervalClaude',
              subtitle: '⚠️ 所有帐号耗尽',
              message: `最早重置: ${earliest.name} (${earliest.remainingMin}m 后)`,
            });
            logFn(`daemon: all accounts exhausted, earliest reset: ${earliest.name} in ${earliest.remainingMin}m`);
            state.allExhaustedNotified = true; // 内存标志，下次成功后清掉
          }
        }
      }
      await interruptibleSleep(checkIntervalMs, shouldStop);
      continue;
    }

    // 执行动作
    for (const act of actions) {
      if (shouldStop()) break;
      const target = act.account;
      const tsIso = nowFn().toISOString();
      if (act.type === ACTION_PING) {
        logFn(`[${tsIso}] schedule: PING ${target.name} (health=${Math.round(health(target, config, now))})`);
        try {
          const result = await pingFn(target, config.ping_prompt);
          if (result.success) {
            await recordPing(target.name, nowFn().toISOString());
            logFn(`[${tsIso}] PING ${target.name}: OK`);
            await refreshUsageAndToken(target.name, logFn);
          } else if (result.limitReached) {
            // 标记 limit_reached（仅内存，下次重读 config 会丢，无所谓）
            target.limit_reached_at = now;
            logFn(`[${tsIso}] PING ${target.name}: LIMIT_REACHED`);
          } else {
            logFn(`[${tsIso}] PING ${target.name}: FAIL code=${result.code}`);
          }
        } catch (err) {
          logFn(`[${tsIso}] PING ${target.name}: ERROR ${err?.message ?? err}`);
        }
      } else if (act.type === ACTION_USE) {
        logFn(`[${tsIso}] schedule: USE ${target.name}`);
        try {
          await performUseFromDaemon(target.name);
          if (config.scheduler.notify !== false) {
            await sendNotification({
              title: 'intervalClaude',
              subtitle: '已切换帐号',
              message: `已切换到 ${target.name}`,
            });
          }
          state.allExhaustedNotified = false;
        } catch (err) {
          logFn(`[${tsIso}] USE ${target.name} 失败: ${err?.message ?? err}`);
        }
      }
    }

    if (shouldStop()) break;
    await interruptibleSleep(checkIntervalMs, shouldStop);
  }

  logFn('daemon: 主循环退出');
}

// 工具函数
function findEarliestReset(accounts) {
  const now = Date.now();
  let best = null;
  for (const a of accounts) {
    const ts = a.last_usage?.five_hour?.resets_at;
    if (!ts) continue;
    const remainMin = (new Date(ts).getTime() - now) / 60_000;
    if (remainMin <= 0) continue;
    if (!best || remainMin < best.remainingMin) {
      best = { name: a.name, remainingMin: Math.round(remainMin) };
    }
  }
  return best;
}

// daemon 内调用 use（复用 use 命令的核心逻辑）
async function performUseFromDaemon(name) {
  const useCommand = (await import('./commands/use.js')).default;
  // use 命令内部会 saveConfig，所以这里不重复
  await useCommand([name]);
}

const state = { allExhaustedNotified: false };  // module-level 状态
```

```js
// src/notifier.js
// macOS 系统通知 (via osascript)
// 仅 darwin 平台生效，其他平台静默 no-op

import { spawn } from 'node:child_process';

/**
 * 转义字符串以嵌入 AppleScript 字符串字面量
 */
export function escapeForApplescript(s) {
  if (s == null) return '';
  return String(s).replace(/\\/g, '\\\\').replace(/"/g, '\\"');
}

/**
 * 构造 osascript 命令参数
 */
export function buildOsascriptArgs({ title, subtitle, message, sound }) {
  const parts = [`display notification "${escapeForApplescript(message)}"`];
  if (title) parts.push(`with title "${escapeForApplescript(title)}"`);
  if (subtitle) parts.push(`subtitle "${escapeForApplescript(subtitle)}"`);
  if (sound) parts.push(`sound name "${escapeForApplescript(sound)}"`);
  return ['-e', parts.join(' ')];
}

/**
 * 发送系统通知
 * @returns {Promise<boolean>} - 成功 true，失败/不支持平台 false
 */
export async function sendNotification({ title, subtitle, message, sound = 'Submarine' }) {
  if (process.platform !== 'darwin') return false;
  return new Promise((resolve) => {
    const args = buildOsascriptArgs({ title, subtitle, message, sound });
    const child = spawn('osascript', args, { stdio: 'ignore' });
    child.on('error', () => resolve(false));
    child.on('exit', (code) => resolve(code === 0));
  });
}
```

```js
function fmtStateLabel(a, activeAccessToken, daemonCache, now) {
  const isActive = activeAccessToken && getAccessToken(a) === activeAccessToken;
  const u = a.last_usage?.five_hour;
  const resetMs = u?.resets_at ? new Date(u.resets_at).getTime() : null;
  const isExhausted = u && u.utilization >= 1.0;
  const isWindowStarted = !!u?.resets_at;
  const isWindowExpired = resetMs && resetMs <= now;

  if (isActive) return '🟢 活跃';
  if (isExhausted) return '🔴 耗尽';
  if (isWindowStarted && !isWindowExpired) return '🔵 备用';
```

```js
function fmtNextPing(account, status, configCache, activeAcc, now) {
  // 活跃帐号
  if (account === activeAcc) return '-';

  const u = account.last_usage?.five_hour;
  // 已激活备用
  if (u?.resets_at && new Date(u.resets_at).getTime() > now) {
    return '-';  // 已经在跑了
  }

  // 未激活：预测下次预 ping 时间
  if (!activeAcc || !configCache) return '-';
  const N = configCache.accounts.length;
  const stagger = configCache.scheduler?.stagger_min ?? Math.round(300 / N);

  // 找未激活帐号按健康度排序的位置
  const dormant = configCache.accounts
    .filter((a) => a !== activeAcc && !a.last_usage?.five_hour?.resets_at)
    .sort((a, b) => (b._health ?? 0) - (a._health ?? 0));
  const idx = dormant.indexOf(account);
  if (idx < 0) return '-';

  // 已激活备用数
  const running = configCache.accounts.filter((a) =>
    a !== activeAcc &&
    a.last_usage?.five_hour?.resets_at &&
    new Date(a.last_usage.five_hour.resets_at).getTime() > now
  ).length;

  const targetIdx = idx + running;
  const targetTimeMin = (targetIdx + 1) * stagger;

  const startMs = status?.startedAt ? new Date(status.startedAt).getTime() : now;
  const elapsedMin = Math.max(0, (now - startMs) / 60_000);
  const remainMin = Math.max(0, targetTimeMin - elapsedMin);
  return `ping@${Math.round(remainMin)}m`;
}
```
